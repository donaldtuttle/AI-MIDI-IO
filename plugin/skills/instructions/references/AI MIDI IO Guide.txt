4. Modularized MIDI Generation

Function to Generate MIDI Files

def generate_midi(notes, tempo, output_path):
"""Generate a MIDI file from a list of notes."""
from MIDIFile import MIDIFile

midi = MIDIFile(1)
track = 0
time = 0
midi.addTempo(track, time, tempo)
for i, note in enumerate(notes):
midi.addNote(
track=track,
channel=0,
pitch=note,
time=time + i,
duration=1,
volume=100
)
save_midi(midi, output_path)

def save_midi(midi, output_path):
"""Save the MIDI file to the specified path with error handling."""
try:
with open(output_path, "wb") as output_file:
midi.writeFile(output_file)
print(f"MIDI file saved to {output_path}")
except Exception as e:
print(f"Error saving MIDI file: {e}")

5. Modularized WAV Generation with Multiple Waveforms

Function to Generate WAV Files

import math
import struct

def generate_wav(frequency, duration, output_path, waveform="sine"):
"""Generate a WAV file with the specified waveform."""
sample_rate = 44100 # Hz
samples = []
for n in range(int(sample_rate * duration)):
t = n / sample_rate
if waveform == "sine":
value = math.sin(2 * math.pi * frequency * t)
elif waveform == "square":
value = 1.0 if math.sin(2 * math.pi * frequency * t) >= 0 else -1.0
elif waveform == "triangle":
value = 2 * abs(2 * (t * frequency - math.floor(t * frequency + 0.5))) - 1
elif waveform == "sawtooth":
value = 2 * (t * frequency - math.floor(t * frequency + 0.5))
else:
raise ValueError(f"Unsupported waveform: {waveform}")
samples.append(value)
byte_data = bytearray()
for sample in samples:
int_value = int(sample * 32767.0)
byte_data.extend(struct.pack('<h', int_value))
save_wav(byte_data, sample_rate, output_path)

def save_wav(byte_data, sample_rate, output_path):
"""Save the WAV file to the specified path with error handling."""
try:
with open(output_path, "wb") as f:
# Write WAV file header
f.write(b'RIFF')
f.write(struct.pack('<I', 36 + len(byte_data))) # Chunk size
f.write(b'WAVE')
f.write(b'fmt ')
f.write(struct.pack('<I', 16)) # Subchunk1Size
f.write(struct.pack('<H', 1)) # AudioFormat
f.write(struct.pack('<H', 1)) # NumChannels
f.write(struct.pack('<I', sample_rate)) # SampleRate
f.write(struct.pack('<I', sample_rate * 2)) # ByteRate
f.write(struct.pack('<H', 2)) # BlockAlign
f.write(struct.pack('<H', 16)) # BitsPerSample
f.write(b'data')
f.write(struct.pack('<I', len(byte_data))) # Subchunk2Size
f.write(byte_data)
print(f"WAV file saved to {output_path}")
except Exception as e:
print(f"Error saving WAV file: {e}")

Example Usage

9. Batch Processing for MIDI and WAV Generation

Batch MIDI Generation

def batch_generate_midi(scales, tempo, directory):
"""Generate multiple MIDI files from a list of scales."""
for i, scale in enumerate(scales):
output_path = os.path.join(directory, f"scale_{i+1}.mid")
generate_midi(scale, tempo, output_path)

# Example Usage
scales = [
[60, 62, 64, 65, 67, 69, 71, 72], # C Major Scale
[62, 64, 66, 67, 69, 71, 73, 74], # D Major Scale
]
batch_generate_midi(scales, 120, MIDI_DIR)

Batch WAV Generation

def batch_generate_wav(frequencies, duration, directory, waveform="sine"):
"""Generate multiple WAV files for a list of frequencies."""
for freq in frequencies:
output_path = os.path.join(directory, f"tone_{int(freq)}Hz.wav")
generate_wav(freq, duration, output_path, waveform)

# Example Usage
frequencies = [440, 493.88, 523.25] # A4, B4, C5
batch_generate_wav(frequencies, 2, WAV_DIR, waveform="sawtooth")
