Dynamic Harmonic Modulation Example:

from mido import MidiFile, MidiTrack, Message
import numpy as np, wave

def generate_dynamic_midi():
    mid = MidiFile()
    track = MidiTrack()
    mid.tracks.append(track)

    # Add MIDI messages with dynamic velocity modulation
    for i, note in enumerate([60, 62, 64, 65, 67, 69, 71, 72]):
        velocity = int(64 + (i * 8)) % 127  # Dynamic scaling
        track.append(Message('note_on', note=note, velocity=velocity, time=120))
        track.append(Message('note_off', note=note, velocity=velocity, time=240))

    mid.save('/mnt/data/midi_generation/dynamic_midi.mid')
    print("MIDI file generated with dynamic modulation.")
